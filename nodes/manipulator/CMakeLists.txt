if(AVIATOR_COMMUNICATION_ONLY OR NOT AVIATOR_BUILD_ROBOTICS_LIBS)
  return()
endif()
add_library(aviator_device STATIC backend_factory.cpp)
target_include_directories(aviator_device PUBLIC ${PROJECT_SOURCE_DIR}/common/include)
target_link_libraries(aviator_device PUBLIC aviator_robotics_deps Threads::Threads)
if(AVIATOR_BUILD_MUJOCO)
  target_sources(aviator_device PRIVATE DataLink_direct.cpp)
  target_link_libraries(aviator_device PUBLIC mujoco_imported)
  target_compile_definitions(aviator_device PUBLIC AVIATOR_HAVE_MUJOCO=1)
endif()
if(AVIATOR_BUILD_XCORE_SDK)
  add_library(aviator_manipulator_sdk SHARED rokae_sdk.cpp)
  target_include_directories(aviator_manipulator_sdk PRIVATE ${AVIATOR_VENDOR_DIR}/xCoreSDK-0.7.1/include)
  target_link_libraries(aviator_manipulator_sdk PRIVATE Eigen3::Eigen Rokae::Rokae Threads::Threads ${CMAKE_DL_LIBS})
  target_link_options(aviator_manipulator_sdk PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic")
  target_sources(aviator_device PRIVATE DataLink_rokae.cpp)
  target_link_libraries(aviator_device PRIVATE aviator_manipulator_sdk)
  target_compile_definitions(aviator_device PUBLIC AVIATOR_HAVE_ROKAE=1)
  install(TARGETS aviator_manipulator_sdk LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
add_executable(manipulator main.cpp)
target_link_libraries(manipulator PRIVATE aviator_device aviator_motion_protocol)
if(AVIATOR_BUILD_MUJOCO AND AVIATOR_BUILD_VIEWER)
  target_sources(manipulator PRIVATE Viewer.cpp)
  target_link_libraries(manipulator PRIVATE glfw)
  target_compile_definitions(manipulator PRIVATE AVIATOR_HAVE_GLFW=1)
endif()
install(TARGETS manipulator RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
