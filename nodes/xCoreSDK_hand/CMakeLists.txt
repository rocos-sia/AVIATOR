find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(xCoreSDK_hand launcher.cpp)
target_compile_definitions(xCoreSDK_hand PRIVATE
  HAND_PYTHON="${Python3_EXECUTABLE}"
  HAND_SOURCE_SCRIPT="${CMAKE_CURRENT_SOURCE_DIR}/main.py")
if(AVIATOR_BUILD_XCORE_SDK AND TARGET Rokae::Rokae)
  add_library(aviator_xcore_hand_sdk SHARED sdk.cpp)
  target_include_directories(aviator_xcore_hand_sdk PRIVATE ${AVIATOR_VENDOR_DIR}/xCoreSDK-0.7.1/include)
  target_link_libraries(aviator_xcore_hand_sdk PRIVATE Eigen3::Eigen Rokae::Rokae Threads::Threads ${CMAKE_DL_LIBS})
  target_link_options(aviator_xcore_hand_sdk PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic")
  add_dependencies(xCoreSDK_hand aviator_xcore_hand_sdk)
  install(TARGETS aviator_xcore_hand_sdk LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
install(TARGETS xCoreSDK_hand RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES main.py transport.py scheduler.py policy.py README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/aviator/xCoreSDK_hand)
if(BUILD_TESTING)
  add_library(xcore_hand_fake_sdk SHARED sdk.cpp)
  target_include_directories(xcore_hand_fake_sdk PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests/fake_sdk)
  add_test(NAME xcore_hand_transport COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_transport.py $<TARGET_FILE:xcore_hand_fake_sdk>)
  add_test(NAME xcore_hand_scheduler COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_scheduler.py)
  add_test(NAME xcore_hand_policy COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_policy.py)
  add_test(NAME xcore_hand_grip_demo COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/examples/xCoreSDK_RH56FTP_485/test_grip_demo.py)
  set_tests_properties(xcore_hand_grip_demo PROPERTIES TIMEOUT 30)
  set_tests_properties(xcore_hand_scheduler PROPERTIES TIMEOUT 30)
  if(TARGET aviator_xcore_hand_sdk)
    add_test(NAME xcore_hand_sdk_initialization COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/test_sdk_initialization.py $<TARGET_FILE:aviator_xcore_hand_sdk>)
    set_tests_properties(xcore_hand_sdk_initialization PROPERTIES TIMEOUT 25)
  endif()
endif()
