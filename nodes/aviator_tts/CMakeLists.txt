find_package(yaml-cpp REQUIRED)
add_library(aviator_tts_rules tts.cpp)
target_include_directories(aviator_tts_rules PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(aviator_tts_rules PRIVATE AVIATOR_TTS_CONFIG="${PROJECT_SOURCE_DIR}/config/tts.yaml")
target_link_libraries(aviator_tts_rules PUBLIC aviator_runtime yaml-cpp)
add_executable(aviator_tts main.cpp)
target_link_libraries(aviator_tts PRIVATE aviator_tts_rules aviator_transport)
install(TARGETS aviator_tts RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY ${PROJECT_SOURCE_DIR}/assets/tts/ DESTINATION ${CMAKE_INSTALL_DATADIR}/aviator/assets/tts)
install(PROGRAMS ${PROJECT_SOURCE_DIR}/tools/tts/generate.py DESTINATION ${CMAKE_INSTALL_DATADIR}/aviator/tools/tts)
install(FILES ${PROJECT_SOURCE_DIR}/tools/tts/requirements.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/aviator/tools/tts)
install(PROGRAMS ${PROJECT_SOURCE_DIR}/scripts/generate_tts.sh DESTINATION ${CMAKE_INSTALL_DATADIR}/aviator/scripts)
