cmake_minimum_required(VERSION 3.16)
project(flight_simulator VERSION 1.0.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This project uses Linux tty and signalfd APIs")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(SDL2 2.0.14 REQUIRED)
find_package(Freetype REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
add_library(simulator_lib STATIC src/config.cpp src/input.cpp src/input_help.cpp
  src/sdl_input.cpp src/rs422.cpp src/serial_port.cpp src/simulator.cpp)
target_include_directories(simulator_lib PUBLIC src)
target_link_libraries(simulator_lib PUBLIC SDL2::SDL2 Freetype::Freetype yaml-cpp Threads::Threads)
target_compile_options(simulator_lib PRIVATE -Wall -Wextra -Wpedantic)
add_executable(flight_simulator src/main.cpp)
target_link_libraries(flight_simulator PRIVATE simulator_lib)
configure_file(config/config.yaml config.yaml COPYONLY)
install(TARGETS flight_simulator RUNTIME DESTINATION bin)
install(FILES config/config.yaml DESTINATION share/flight_simulator)
install(FILES README.md PROTOCOL.md LICENSE DESTINATION share/flight_simulator)
include(CTest)
if(BUILD_TESTING)
  add_executable(protocol_test tests/protocol_test.cpp)
  target_link_libraries(protocol_test PRIVATE simulator_lib)
  add_test(NAME protocol COMMAND protocol_test)
  add_executable(serial_test tests/serial_test.cpp)
  target_link_libraries(serial_test PRIVATE simulator_lib)
  target_link_options(serial_test PRIVATE "-Wl,--wrap=tcsetattr")
  add_test(NAME serial COMMAND serial_test)
  add_test(NAME help COMMAND flight_simulator --help)
  add_executable(simulator_driver tests/simulator_driver.cpp)
  target_link_libraries(simulator_driver PRIVATE simulator_lib)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME dual_pty COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/dual_pty_test.py
      $<TARGET_FILE:simulator_driver> ${CMAKE_CURRENT_SOURCE_DIR}/config/config.yaml)
    set_tests_properties(dual_pty PROPERTIES TIMEOUT 15)
    add_test(NAME config_validation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/config_test.py
      $<TARGET_FILE:flight_simulator> ${CMAKE_CURRENT_SOURCE_DIR}/config/config.yaml)
  endif()
endif()
set(CPACK_GENERATOR TGZ)
set(CPACK_SOURCE_GENERATOR TGZ)
set(CPACK_PACKAGE_FILE_NAME "flight_simulator-${PROJECT_VERSION}-${CMAKE_SYSTEM_PROCESSOR}-linux")
set(CPACK_SOURCE_PACKAGE_FILE_NAME "flight_simulator-${PROJECT_VERSION}-source")
set(CPACK_SOURCE_IGNORE_FILES "/build[^/]*/;/[.]git/;__pycache__;[.]pyc$;[.]tar[.]gz$;~$")
include(CPack)
