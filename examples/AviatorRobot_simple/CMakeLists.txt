cmake_minimum_required(VERSION 3.22)
project(AviatorRobotSimple LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(AVIATOR_WITH_SIMULATION "Build MuJoCo backend and simulation tests" ON)
option(AVIATOR_WITH_ROKAE "Build xCore SDK backend (does not connect during build/tests)" ON)
option(AVIATOR_WITH_VIEWER "Build GLFW viewer" ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
# DT_RPATH is transitive, so the bundled indirect dependencies also resolve locally.
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
set(CMAKE_BUILD_RPATH "$ORIGIN/../lib;$ORIGIN/../dependencies/lib")
set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib")
add_link_options("-Wl,--disable-new-dtags" "-Wl,-z,defs")
find_package(Threads REQUIRED)
include(CTest)
include(cmake/Dependencies.cmake)

add_library(aviator_control SHARED src/Aviator.cpp src/backend_factory.cpp
    src/Kinematics_pin_ik.cpp src/CollisionChecker_pinocchio.cpp)
target_include_directories(aviator_control PUBLIC include PRIVATE src)
target_link_libraries(aviator_control PUBLIC aviator_dependencies Threads::Threads)
if(AVIATOR_WITH_SIMULATION)
    add_library(mujoco SHARED IMPORTED)
    set_target_properties(mujoco PROPERTIES IMPORTED_LOCATION "${AVIATOR_DEPS}/lib/libmujoco.so")
    target_sources(aviator_control PRIVATE src/DataLink_direct.cpp)
    target_link_libraries(aviator_control PRIVATE mujoco)
    target_compile_definitions(aviator_control PRIVATE AVIATOR_HAVE_MUJOCO=1)
endif()
if(AVIATOR_WITH_ROKAE)
    # Keep SDK's embedded KDL private in a separate DSO; the bridge contains no KDL types.
    add_library(aviator_rokae_sdk SHARED src/rokae_sdk.cpp)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
        message(FATAL_ERROR "Vendored xCore SDK targets Linux x86_64; disable AVIATOR_WITH_ROKAE on other platforms")
    endif()
    target_include_directories(aviator_rokae_sdk PRIVATE
        "${AVIATOR_VENDOR}/xcore/include")
    target_link_libraries(aviator_rokae_sdk PRIVATE Eigen3::Eigen
        "${AVIATOR_VENDOR}/xcore/lib/linux-x86_64/libxCoreSDK.a"
        "${AVIATOR_VENDOR}/xcore/lib/linux-x86_64/libxMateModel.a"
        Threads::Threads ${CMAKE_DL_LIBS})
    target_link_options(aviator_rokae_sdk PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic")
    target_sources(aviator_control PRIVATE src/DataLink_rokae.cpp)
    target_link_libraries(aviator_control PRIVATE aviator_rokae_sdk)
    target_compile_definitions(aviator_control PUBLIC AVIATOR_HAVE_ROKAE=1)
    install(TARGETS aviator_rokae_sdk LIBRARY DESTINATION lib)
endif()

# One entry point, one demo and one command dispatcher for both backends.
add_executable(aviator src/main.cpp)
target_link_libraries(aviator PRIVATE aviator_control)
if(AVIATOR_WITH_SIMULATION)
    target_link_libraries(aviator PRIVATE mujoco)
    target_compile_definitions(aviator PRIVATE AVIATOR_HAVE_MUJOCO=1)
    if(AVIATOR_WITH_VIEWER)
        add_library(aviator_viewer STATIC src/Viewer.cpp)
        add_dependencies(aviator_viewer source_mujoco)
        target_include_directories(aviator_viewer PUBLIC "${AVIATOR_DEPS}/include")
        target_link_libraries(aviator_viewer PUBLIC mujoco glfw)
        target_link_libraries(aviator PRIVATE aviator_viewer)
        target_compile_definitions(aviator PRIVATE AVIATOR_HAVE_GLFW=1)
    endif()
endif()
add_custom_target(aviator_assets ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${PROJECT_SOURCE_DIR}/config" "${CMAKE_BINARY_DIR}/bin/config"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${PROJECT_SOURCE_DIR}/model" "${CMAKE_BINARY_DIR}/bin/model")
add_dependencies(aviator aviator_assets)
install(TARGETS aviator aviator_control RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
install(DIRECTORY config model DESTINATION bin)
# Do not ship an obsolete library left in a dependency prefix by an older build.
install(DIRECTORY "${AVIATOR_DEPS}/lib/" DESTINATION lib FILES_MATCHING
    PATTERN "*.so*" PATTERN "liborocos-kdl.so*" EXCLUDE)
install(DIRECTORY "${AVIATOR_VENDOR}/licenses/" DESTINATION share/aviator/licenses)

if(BUILD_TESTING)
    add_executable(aviator_pose_test tests/test_pose.cpp)
    target_include_directories(aviator_pose_test PRIVATE include)
    target_link_libraries(aviator_pose_test PRIVATE aviator_dependencies)
    add_test(NAME aviator_pose COMMAND aviator_pose_test)

    add_executable(aviator_pin_ik_test third_party/pin_ik/tests/standalone_tests.cpp)
    target_link_libraries(aviator_pin_ik_test PRIVATE aviator_dependencies)
    add_test(NAME aviator_pin_ik COMMAND aviator_pin_ik_test "${AVIATOR_VENDOR}/pin_ik/tests/robot.urdf")
    set_tests_properties(aviator_pin_ik PROPERTIES TIMEOUT 60 ENVIRONMENT "LD_BIND_NOW=1")
endif()
if(BUILD_TESTING AND AVIATOR_WITH_SIMULATION)
    foreach(_test basic acceptance backend_selection realtime move_wheel servo_wheel kinematics)
        add_executable(aviator_${_test}_test "tests/test_${_test}.cpp")
        target_link_libraries(aviator_${_test}_test PRIVATE aviator_control mujoco)
        target_compile_definitions(aviator_${_test}_test PRIVATE
            AVIATOR_CONFIG_DIR="${PROJECT_SOURCE_DIR}/config"
            AVIATOR_MODEL_DIR="${PROJECT_SOURCE_DIR}/model")
        add_test(NAME aviator_${_test} COMMAND aviator_${_test}_test)
        set_tests_properties(aviator_${_test} PROPERTIES TIMEOUT 600 ENVIRONMENT "LD_BIND_NOW=1")
    endforeach()
    add_test(NAME aviator_demo COMMAND aviator --demo --headless)
    set_tests_properties(aviator_demo PROPERTIES TIMEOUT 600 ENVIRONMENT "LD_BIND_NOW=1")
    add_test(NAME aviator_servo_demo COMMAND aviator --servo-demo --headless)
    set_tests_properties(aviator_servo_demo PROPERTIES TIMEOUT 60 ENVIRONMENT "LD_BIND_NOW=1")
endif()
if(BUILD_TESTING)
    add_executable(aviator_open_loop_test tests/test_open_loop.cpp)
    target_include_directories(aviator_open_loop_test PRIVATE include)
    add_test(NAME aviator_open_loop COMMAND aviator_open_loop_test)
endif()
if(BUILD_TESTING)
    add_executable(aviator_controller_safety_test tests/test_controller_safety.cpp)
    target_link_libraries(aviator_controller_safety_test PRIVATE aviator_control)
    target_compile_definitions(aviator_controller_safety_test PRIVATE
        AVIATOR_CONFIG_DIR="${PROJECT_SOURCE_DIR}/config")
    add_test(NAME aviator_controller_safety COMMAND aviator_controller_safety_test)
    if(AVIATOR_WITH_SIMULATION)
        find_package(Python3 COMPONENTS Interpreter QUIET)
        if(Python3_Interpreter_FOUND)
            add_test(NAME aviator_interactive COMMAND ${Python3_EXECUTABLE}
                "${PROJECT_SOURCE_DIR}/tests/test_interactive.py" "$<TARGET_FILE:aviator>")
            set_tests_properties(aviator_interactive PROPERTIES TIMEOUT 30)
        endif()
    endif()
endif()
# Run manually on a display; keep default CTest usable on headless build machines.
if(BUILD_TESTING AND AVIATOR_WITH_SIMULATION AND AVIATOR_WITH_VIEWER)
    add_executable(aviator_viewer_test tests/test_viewer.cpp)
    target_link_libraries(aviator_viewer_test PRIVATE aviator_viewer)
    target_compile_definitions(aviator_viewer_test PRIVATE AVIATOR_MODEL_DIR="${PROJECT_SOURCE_DIR}/model")
endif()
