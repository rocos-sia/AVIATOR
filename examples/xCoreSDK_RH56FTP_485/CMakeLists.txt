cmake_minimum_required(VERSION 3.16)
project(xCoreSDK_RH56FTP_485 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
    message(FATAL_ERROR "This example supports Linux x86_64")
endif()
set(XCORE_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/xCoreSDK-0.7.1" CACHE PATH "xCoreSDK 0.7.1.ar_10 root")
foreach(lib IN ITEMS xCoreSDK xMateModel)
    set(${lib}_LIBRARY "${XCORE_SDK_ROOT}/lib/Linux/x86_64/lib${lib}.a")
    if(NOT EXISTS "${${lib}_LIBRARY}")
        message(FATAL_ERROR "Missing ${${lib}_LIBRARY}; set XCORE_SDK_ROOT to a complete SDK")
    endif()
endforeach()
find_package(Threads REQUIRED)
find_package(Qt5 5.12 REQUIRED COMPONENTS Widgets)
add_library(hand_backend src/backend.cpp)
target_include_directories(hand_backend PUBLIC src)
target_include_directories(hand_backend SYSTEM PRIVATE "${XCORE_SDK_ROOT}/include" "${XCORE_SDK_ROOT}/external")
target_compile_definitions(hand_backend PRIVATE EIGEN_DONT_ALIGN_STATICALLY)
target_link_libraries(hand_backend PRIVATE "${xCoreSDK_LIBRARY}" "${xMateModel_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
add_library(hand_window src/window.cpp)
target_link_libraries(hand_window PUBLIC hand_backend Qt5::Widgets)
add_executable(xCoreSDK_RH56FTP_485 src/main.cpp)
target_link_libraries(xCoreSDK_RH56FTP_485 PRIVATE hand_window)
add_executable(rh56_485_probe src/probe.cpp)
target_link_libraries(rh56_485_probe PRIVATE hand_window)
target_include_directories(rh56_485_probe SYSTEM PRIVATE "${XCORE_SDK_ROOT}/include" "${XCORE_SDK_ROOT}/external")
target_compile_definitions(rh56_485_probe PRIVATE EIGEN_DONT_ALIGN_STATICALLY)
configure_file(config.ini config.ini COPYONLY)
foreach(target IN ITEMS hand_backend hand_window xCoreSDK_RH56FTP_485 rh56_485_probe)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
include(CTest)
