cmake_minimum_required(VERSION 3.16)
project(rh56_sdk_demo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
    message(FATAL_ERROR "This demo requires Linux x86_64")
endif()
set(XCORE_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/xCoreSDK-0.7.1" CACHE PATH "xCoreSDK 0.7.1.ar_10 directory")
find_package(Threads REQUIRED)
add_executable(rh56_sdk_demo main.cpp)
target_include_directories(rh56_sdk_demo SYSTEM PRIVATE "${XCORE_SDK_ROOT}/include" "${XCORE_SDK_ROOT}/external")
target_compile_definitions(rh56_sdk_demo PRIVATE EIGEN_DONT_ALIGN_STATICALLY)
target_compile_options(rh56_sdk_demo PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(rh56_sdk_demo PRIVATE
    "${XCORE_SDK_ROOT}/lib/Linux/x86_64/libxCoreSDK.a"
    "${XCORE_SDK_ROOT}/lib/Linux/x86_64/libxMateModel.a"
    Threads::Threads ${CMAKE_DL_LIBS})
include(CTest)
if(BUILD_TESTING)
    add_test(NAME init_485_dry_run COMMAND rh56_sdk_demo --init-485 192.168.1.160 --dry-run)
    set_tests_properties(init_485_dry_run PROPERTIES PASS_REGULAR_EXPRESSION "xPanelRS485Init.*setxPanelRS485Overtime\\(500\\)")
    add_executable(protocol_test protocol_test.cpp)
    add_test(NAME rh56_protocol COMMAND protocol_test)
    add_test(NAME read_dry_run COMMAND rh56_sdk_demo 192.168.1.160 --dry-run)
    set_tests_properties(read_dry_run PROPERTIES
        PASS_REGULAR_EXPRESSION "eb 90 01 04 11 0a 06 0c 32"
        FAIL_REGULAR_EXPRESSION "setxPanelRS485Overtime")
    add_test(NAME read_id2_dry_run COMMAND rh56_sdk_demo 192.168.1.160 2 --dry-run)
    set_tests_properties(read_id2_dry_run PROPERTIES PASS_REGULAR_EXPRESSION "eb 90 02 04 11 0a 06 0c 33")
    add_test(NAME reject_write COMMAND rh56_sdk_demo 192.168.1.160 1 900 900 900 900 900 900 --dry-run)
    add_test(NAME reject_bad_id COMMAND rh56_sdk_demo 192.168.1.160 255 --dry-run)
    add_test(NAME reject_malformed_id COMMAND rh56_sdk_demo 192.168.1.160 1x --dry-run)
    set_tests_properties(reject_write reject_bad_id reject_malformed_id PROPERTIES WILL_FAIL TRUE)
endif()
