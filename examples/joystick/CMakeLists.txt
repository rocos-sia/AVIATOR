cmake_minimum_required(VERSION 3.16)
project(usb_joystick LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This example requires Linux /dev/input/js*.")
endif()
add_library(joystick STATIC src/joystick.cpp)
target_include_directories(joystick PUBLIC include)
target_compile_features(joystick PUBLIC cxx_std_17)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/index.html" JOYSTICK_HTML)
configure_file(src/page.hpp.in generated/page.hpp @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS index.html)
add_executable(joystick_web src/main.cpp)
target_include_directories(joystick_web PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(joystick_web PRIVATE joystick)

option(JOYSTICK_BUILD_TESTS "Build joystick regression tests" OFF)
if(JOYSTICK_BUILD_TESTS)
    enable_testing()
    add_executable(joystick_json_test tests/json.cpp)
    target_include_directories(joystick_json_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(joystick_json_test PRIVATE joystick)
    add_test(NAME joystick_json_utf8 COMMAND joystick_json_test)
    find_program(JOYSTICK_NODE node)
    if(JOYSTICK_NODE)
        add_test(NAME joystick_button COMMAND "${JOYSTICK_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/button.cjs")
    endif()
endif()
